vẽ cây cấu trúc thư mục cho dự án
function Draw-Tree {
    param(
        [string]$Path,
        [string]$Prefix = "",
        [string[]]$ExcludeDirs = @("venv",".venv",".env","__pycache__",".vscode",".pytest_cache",".git")
    )

    $items = Get-ChildItem -Path $Path -Force | Where-Object {
        -not ($_.Name -like "*.pyc" -or $_.Name -like "*.db") -and
        -not ($ExcludeDirs -contains $_.Name)
    }

    for ($i = 0; $i -lt $items.Count; $i++) {
        $item = $items[$i]
        $isLast = ($i -eq $items.Count - 1)
        $connector = if ($isLast) { "└── " } else { "├── " }

        Write-Output ($Prefix + $connector + $item.Name)

        if ($item.PSIsContainer) {
            $newPrefix = if ($isLast) { $Prefix + "    " } else { $Prefix + "│   " }
            Draw-Tree -Path $item.FullName -Prefix $newPrefix -ExcludeDirs $ExcludeDirs
        }
    }
}
